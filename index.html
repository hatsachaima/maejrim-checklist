<script>

// ================= CONFIGURATION (ตั้งค่าตรงนี้เรียบร้อยแล้ว) =================

const GOOGLE_FORM_URL = "https://docs.google.com/forms/d/e/1FAIpQLSf2cwVSZwV7_5bVWt2ESdJpC99DAksfysFV0DXaX3l1DgHgng/formResponse"; 

const ENTRY_TASK_ID = "entry.2076841301";   

const ENTRY_STATUS = "entry.1858284539";    

// ===============================================================


const KEY = 'maejrim_tasks_v1.1';


function loadState() {

  try {

    return JSON.parse(localStorage.getItem(KEY) || '{}');

  } catch(e) {

    return {};

  }

}


function saveState(state) {

  try {

    localStorage.setItem(KEY, JSON.stringify(state));

  } catch(e) {}

}


function updateUI() {

  const state = loadState();

  document.querySelectorAll('.check-btn').forEach(function(btn) {

    const id = btn.id;

    if (state[id]) {

      btn.classList.add('done');

      btn.innerText = "✓ เสร็จ";

    } else {

      btn.classList.remove('done');

      btn.innerText = "ทำแล้ว";

    }

  });

}


function toggleTask(id) {

  const state = loadState();

  state[id] = !state[id];

  const currentStatus = state[id] ? "done" : "undone";

  

  saveState(state);

  updateUI();

  

  const indicator = document.getElementById('syncIndicator');

  indicator.innerText = "กำลังส่งข้อมูลไปพัทลุง...";

  indicator.style.color = "#d48806";

  

  const formData = new FormData();

  formData.append(ENTRY_TASK_ID, id);

  formData.append(ENTRY_STATUS, currentStatus);

  

  fetch(GOOGLE_FORM_URL, {

    method: "POST",

    body: formData,

    mode: "no-cors"

  }).then(() => {

    indicator.innerText = "✓ บันทึกข้อมูลส่งพัทลุงสำเร็จ";

    indicator.style.color = "#389e0d";

  }).catch((error) => {

    console.error("Error syncing with Google Sheets:", error);

    indicator.innerText = "❌ เน็ตหลุด! แต่บันทึกในเครื่องแล้ว";

    indicator.style.color = "#cf1322";

  });

}


function resetAll() {

  if (!confirm('คุณป้าต้องการล้างรายการทั้งหมดเพื่อเริ่มวันใหม่ใช่ไหมครับ?')) return;

  

  const indicator = document.getElementById('syncIndicator');

  indicator.innerText = "กำลังส่งคำสั่งเริ่มวันใหม่...";

  

  const formData = new FormData();

  formData.append(ENTRY_TASK_ID, "ALL_TASKS");

  formData.append(ENTRY_STATUS, "RESET_DAY");

  

  fetch(GOOGLE_FORM_URL, {

    method: "POST",

    body: formData,

    mode: "no-cors"

  }).then(() => {

    try { localStorage.removeItem(KEY); } catch(e) {}

    updateUI();

    indicator.innerText = "✓ เริ่มวันใหม่สำเร็จ (ระบบซิงค์แล้ว)";

    indicator.style.color = "#389e0d";

  }).catch(() => {

    try { localStorage.removeItem(KEY); } catch(e) {}

    updateUI();

    indicator.innerText = "⚠️ ล้างในเครื่องแล้ว แต่เน็ตขัดข้อง";

    indicator.style.color = "#cf1322";

  });

}


window.onload = function() {

  updateUI();

};

</script>
